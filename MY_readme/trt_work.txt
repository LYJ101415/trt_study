🔧 第一步：构建生产级 Engine（关键前置）
在生产部署前，必须重新构建一个融合了 NMS Plugin + Dynamic Shape 的 Engine：
# 使用 trtexec 构建生产级 INT8 Engine
trtexec \
    --onnx=yolov8n_int8_fixed.onnx \
    --saveEngine=yolov8n_production.engine \
    --int8 \
    --fp16 \                          # 允许敏感层自动回退 FP16
    --minShapes=images:1x3x640x640 \  # 动态 batch 下界
    --optShapes=images:8x3x640x640 \  # 最优 batch（根据GPU调优）
    --maxShapes=images:32x3x640x640 \ # 动态 batch 上界
    --buildOnly \
    --verbose

实现 yolo_trt_engine.cpp
#include "yolo_trt_engine.h"
#include <fstream>
#include <iostream>
#include <cassert>

// 自定义 Logger
class TrtLogger : public nvinfer1::ILogger {
    void log(Severity severity, const char* msg) noexcept override {
        if (severity <= Severity::kWARNING)
            std::cerr << "[TRT] " << msg << std::endl;
    }
};
static TrtLogger g_logger;

YoloTrtEngine::YoloTrtEngine(const std::string& engine_path, int max_batch_size)
    : max_batch_(max_batch_size) {

    // 1. 反序列化 Engine
    std::ifstream file(engine_path, std::ios::binary | std::ios::ate);
    if (!file) throw std::runtime_error("Cannot open engine: " + engine_path);
    auto size = file.tellg();
    file.seekg(0);
    std::vector<char> buffer(size);
    file.read(buffer.data(), size);

    runtime_.reset(nvinfer1::createInferRuntime(g_logger));
    engine_.reset(runtime_->deserializeCudaEngine(buffer.data(), size));
    if (!engine_) throw std::runtime_error("Failed to deserialize engine");

    context_.reset(engine_->createExecutionContext());
    if (!context_) throw std::runtime_error("Failed to create execution context");

    // 2. 预分配 GPU 内存（一次性，永不释放）
    input_bytes_ = max_batch_ * 3 * 640 * 640 * sizeof(float);
    // 假设输出为 [batch, 10, 8400] 或 BatchedNMS 后的 [batch, max_det, 7]
    output_bytes_ = max_batch_ * 300 * 7 * sizeof(float);  // 按 NMS Plugin 输出

    cudaMalloc(&d_input_, input_bytes_);
    cudaMalloc(&d_output_, output_bytes_);

    // 3. 绑定静态地址（V3 API）
    context_->setTensorAddress("images", d_input_);
    context_->setTensorAddress("output0", d_output_);  // 或 "num_detections" 等

    std::cout << "[Engine] Loaded. MaxBatch=" << max_batch_
              << ", InputMem=" << input_bytes_ / 1024 << "KB"
              << ", OutputMem=" << output_bytes_ / 1024 << "KB" << std::endl;
}

YoloTrtEngine::~YoloTrtEngine() {
    cudaFree(d_input_);
    cudaFree(d_output_);
}

size_t YoloTrtEngine::getInputSizeBytes() const { return input_bytes_; }

bool YoloTrtEngine::infer(const float* h_input, int batch_size,
                          std::vector<Detection>& results,
                          cudaStream_t stream) {
    if (batch_size > max_batch_) return false;

    // 设置动态 batch shape
    context_->setInputShape("images",
        nvinfer1::Dims4{batch_size, 3, 640, 640});

    // Host → Device（异步）
    cudaMemcpyAsync(d_input_, h_input,
                    batch_size * 3 * 640 * 640 * sizeof(float),
                    cudaMemcpyHostToDevice, stream);

    // 异步推理
    if (!context_->executeAsyncV3(stream)) {
        std::cerr << "[TRT] executeAsyncV3 failed!" << std::endl;
        return false;
    }

    // Device → Host（异步）
    // 注意：如果使用 BatchedNMS Plugin，输出格式不同，需适配
    size_t out_size = batch_size * 300 * 7 * sizeof(float);
    std::vector<float> h_output(out_size / sizeof(float));
    cudaMemcpyAsync(h_output.data(), d_output_, out_size,
                    cudaMemcpyDeviceToHost, stream);

    // 同步等待当前帧完成（生产环境通常用回调/事件代替）
    cudaStreamSynchronize(stream);

    // 解析结果（如果 NMS 已在 Plugin 中完成）
    results.clear();
    for (int b = 0; b < batch_size; ++b) {
        for (int d = 0; d < 300; ++d) {
            int offset = (b * 300 + d) * 7;
            float score = h_output[offset + 4];
            if (score < 0.001f) continue;
            Detection det;
            det.x1 = h_output[offset + 0];
            det.y1 = h_output[offset + 1];
            det.x2 = h_output[offset + 2];
            det.y2 = h_output[offset + 3];
            det.score = score;
            det.class_id = static_cast<int>(h_output[offset + 5]);
            results.push_back(det);
        }
    }
    return true;
}
⚡ 第三步：多流并发调度器（工业核心）
单流无法打满 GPU。生产环境必须使用多 Stream + 请求队列：
class InferenceScheduler {
    std::shared_ptr<YoloTrtEngine> engine_;
    int num_streams_;
    std::vector<cudaStream_t> streams_;
    std::atomic<int> next_stream_{0};

public:
    InferenceScheduler(std::shared_ptr<YoloTrtEngine> engine, int num_streams = 4)
        : engine_(engine), num_streams_(num_streams) {
        streams_.resize(num_streams);
        for (auto& s : streams_) cudaStreamCreate(&s);
    }

    ~InferenceScheduler() {
        for (auto& s : streams_) cudaStreamDestroy(s);
    }

    // Round-Robin 分配 Stream（无锁）
    cudaStream_t acquireStream() {
        int idx = next_stream_.fetch_add(1, std::memory_order_relaxed) % num_streams_;
        return streams_[idx];
    }

    bool submit(const float* input, int batch, std::vector<Detection>& out) {
        auto stream = acquireStream();
        return engine_->infer(input, batch, out, stream);
    }
};
📊 第四步：CUDA 预处理（消除 CPU 瓶颈）
工业部署中 预处理必须在 GPU 上完成，否则 CPU resize+normalize 会吃掉 3~5ms/帧：
// preprocess_kernel.cu
__global__ void letterbox_kernel(
    const uint8_t* __restrict__ src, float* __restrict__ dst,
    int src_h, int src_w, int dst_h, int dst_w,
    float scale, int pad_left, int pad_top)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= dst_w || y >= dst_h) return;

    float val = 114.0f / 255.0f;  // pad value normalized
    int sx = (int)((x - pad_left) / scale);
    int sy = (int)((y - pad_top) / scale);

    if (sx >= 0 && sx < src_w && sy >= 0 && sy < src_h) {
        // BGR→RGB + Normalize，直接从 uint8 转 float
        int src_idx = (sy * src_w + sx) * 3;
        dst[0 * dst_h * dst_w + y * dst_w + x] = src[src_idx + 2] / 255.0f; // R
        dst[1 * dst_h * dst_w + y * dst_w + x] = src[src_idx + 1] / 255.0f; // G
        dst[2 * dst_h * dst_w + y * dst_w + x] = src[src_idx + 0] / 255.0f; // B
    } else {
        dst[0 * dst_h * dst_w + y * dst_w + x] = val;
        dst[1 * dst_h * dst_w + y * dst_w + x] = val;
        dst[2 * dst_h * dst_w + y * dst_w + x] = val;
    }
}

📌 最终建议
不要从零造轮子：如果你的团队没有专职 C++ CUDA 工程师，首选 Triton Inference Server。它封装了上述所有工业最佳实践（动态 Batch、多流、GPU 预处理、健康检查、指标导出），你只需提供 .engine 文件和配置。
如果必须自研 C++：以上代码是完整的生产级骨架，但还需补充：错误处理、日志框架、配置文件解析、Graceful Shutdown、Health Check Endpoint。
INT8 Engine 必须带 FP16 Fallback：你之前发现的 Conv_6 / Conv_250 问题，在生产 Engine 中通过 --fp16 标志让 TRT 自动处理，不要手动排除层。
NMS 必须进 Engine：CPU NMS 在高 QPS 下会成为致命瓶颈。使用 EfficientNMS_TRT 或 BatchedNMS_TRT Plugin。
